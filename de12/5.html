プログラムの説明

1from openai import OpenAI
2import requests#importが二つあるのは最初に両方のAPIを使うための道具を用意している
3
4client = OpenAI(api_key="ChatGPTのapi")
5
6mood = input("今日の気分を入力してください：")#打ってもらった気分をmoodの変数に変換してる
7
8response = client.chat.completions.create(#apiへ指示を送っている
9    model="gpt-6-luna",
10    messages=[
11         #roleは誰からのメッセージなのかを示す　contetはそのメッセージの内容。roleは二つに分けられてsystem → AIの役割や振る舞いを指定する　user → 人間が入力した内容をAIに渡す これをすることで、人間のコメントに対して    aiがただ返答するのではなく、人にアドバイスを送るという信念をもって返答してくれる      
12        {"role": "system", "content": "あなたは人の気持ちに寄り添い、今日一日のアドバイスを提供するアシスタントです。"},
13        {"role": "user", "content": mood}
14    ]
15)
16
17#AIから返ってきたデータの中から、実際の文章だけを取り出して advice という変数に保存する。
18advice = response.choices[0].message.content
19
20quotes = []
21
22#for_in~でquoteを5回繰り返し、上記の[]でリストを作る。
23for _ in range(5):
24    quote_response = requests.get(
25        "https://api.api-ninjas.com/v2/randomquotes?categories=success,wisdom",
26        headers={"X-Api-Key": "quoteのapi"},
27        timeout=30
28    )
29
30    quote_response.raise_for_status()
31    #エラーを処理するときにjsonを使う
32    quote_data = quote_response.json()
33    #append()は上記で作ったリストにデータをぶち込む
34    quotes.append(quote_data[0])
35    
36    quote_candidates = ""
37#iは名言に番号pはその番号に対応する名言の辞書
38for i, q in enumerate(quotes, 1):
39    #candidatesは上の作業で番号を付けた文章を羅列する
40    quote_candidates += f"{i}. {q['quote']} (作者: {q['author']})\n"#\nは改行
41
42
43selection_response = client.chat.completions.create(
44    model="gpt-6-luna",
45    messages=[
46        {
47            "role": "system",
48            "content": "ユーザーの気分を理解し、候補の名言から最も気分に合うものを1つ選んでください。名言の英文と作者をそのまま出力してください。理由や解説は不要です。出力形式は名言の英文、改行、作者名の順にしてください。"
49        },
50        {
51            "role": "user",
52            #moodで人間の気分、candidatesは上記でまとめた名言をaiに渡す
53            "content": f"今日の気分：{mood}\n\n候補の名言：\n{quote_candidates}"
54        }
55    ]
56)
57#aiが選択した名言を受け取る
58selected_quote = selection_response.choices[0].message.content
59#選ばれた名言と作者を保存する
60selected_lines = selected_quote.split("\n")
61
62#リストの0番（名言の英文）をquoteに
63quote = selected_lines[0]
64#リストの一番（作者の情報）をauthorに　replaceは("a","b")aをbに変換する役割　今回は[1]の中にある"作者"の文字を消す　stripは先頭に空白ができたらそれを消す
65author = selected_lines[1].replace("作者:", "").strip()
66
67selected_quote = selection_response.choices[0].message.content
68translation_response = client.chat.completions.create(
69    model="gpt-6-luna",
70    messages=[
71        {"role": "system", "content": "あなたは翻訳者です。入力された英文を自然な日本語に翻訳してください。翻訳結果だけを出力してください。"},
72        #以前は mood を渡していたけど、今回は quote を渡している。つまり、ChatGPTに「今日の気分」ではなく、「取得した英語の名言」を翻訳対象として渡している
73        {"role": "user", "content": quote}
74    ]
75)
76#日本が訳された結果を取り出す
77translated_quote = translation_response.choices[0].message.content
78
79#advice → ChatGPT APIから受け取ったアドバイス quote → 名言APIから受け取った言葉 author → 名言APIから受け取った作者
80#print("==============================")ただ見やすくするための装飾品
81print("\n==============================")
82print("今日のアドバイス")
83print("==============================")
84print(advice)
85
86print("\n今日の言葉")
87print("------------------------------")
88print(quote)
89print("作者:", author)
90print("日本語訳:", translated_quote)

全体の流れとしては、人間が今日の気分を打ち込む
　　　　　　　　　　　　　　　　↓
　　　　　　　　　　Chat GPTが今日一日のアドバイスをくれる
　　　　　　　　　　　　　　　　↓
　　　　　　　　　　aiが名言集のホームページからその日にぴったりな言葉を提示してくれる
aiに"role": "system"を用いて役割を固定化し、毎回適切なアドバイスが送れるようにした。名言は、ランダムに表示されてしまうと今日の気分に適していないことが多くなってしまう。 なのでquoteのapi側に5つ名言を取りだすよう設定し、選ばれた5つの中から、"role":"system"でaiが今のmoodに適した名言を抽出するようにした。しかし、取り出した名言は英語で表示されてしまったため、またもや"role":"system"でaiの役割を和訳者に固定化し、日本語で表示させるようにした。 最後に、これらの過程で設定されたadvice,author,translated_quoteをprintを用いて表示させた。
